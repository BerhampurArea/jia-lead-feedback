<script>
  // ⚠️ Replace with your deployed Apps Script Web App URL (ends in /exec)
  const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbwQpzsrRF8SfVVBPxAREDAM45aKnrLsh6-g5U-CDPNCfB8Kdok2MAj-tOfhwtiXbDpIAg/exec";

  const form = document.getElementById('feedbackForm');
  const callStatusChips = document.querySelectorAll('#callStatusChips .chip');
  const responseChips = document.querySelectorAll('#customerResponseChips .chip');
  const responseSection = document.getElementById('responseSection');
  const bannerError = document.getElementById('bannerError');
  const submitBtn = document.getElementById('submitBtn');
  const spinner = document.getElementById('spinner');
  const submitLabel = document.getElementById('submitLabel');

  let selectedCallStatus = '';
  let selectedResponse = '';

  const needsResponse = (status) => status === 'Called' || status === 'Call but did not connect';

  callStatusChips.forEach(chip => {
    chip.addEventListener('click', () => {
      callStatusChips.forEach(c => c.classList.remove('active'));
      chip.classList.add('active');
      selectedCallStatus = chip.dataset.status;

      if (needsResponse(selectedCallStatus)) {
        responseSection.classList.add('show');
      } else {
        responseSection.classList.remove('show');
        selectedResponse = '';
        responseChips.forEach(c => c.classList.remove('active'));
      }
    });
  });

  responseChips.forEach(chip => {
    chip.addEventListener('click', () => {
      responseChips.forEach(c => c.classList.remove('active'));
      chip.classList.add('active');
      selectedResponse = chip.dataset.resp;
    });
  });

  document.getElementById('mobileNumber').addEventListener('input', (e) => {
    e.target.value = e.target.value.replace(/\D/g, '').slice(0, 10);
  });

  document.getElementById('logAnotherBtn').addEventListener('click', () => {
    document.getElementById('successScreen').style.display = 'none';
    document.getElementById('formCard').style.display = 'block';
    form.reset();
    callStatusChips.forEach(c => c.classList.remove('active'));
    responseChips.forEach(c => c.classList.remove('active'));
    responseSection.classList.remove('show');
    selectedCallStatus = '';
    selectedResponse = '';
    bannerError.style.display = 'none';
  });

  form.addEventListener('submit', async (e) => {
    e.preventDefault();
    bannerError.style.display = 'none';

    const orderId = document.getElementById('orderId').value.trim();
    const riderPrmId = document.getElementById('riderPrmId').value.trim();
    const mobileNumber = document.getElementById('mobileNumber').value.trim();
    const customerName = document.getElementById('customerName').value.trim();
    const territory = document.getElementById('territory').value;
    const telecallerName = document.getElementById('telecallerName').value.trim();
    const remarks = document.getElementById('remarks').value.trim();

    let valid = true;
    if (!orderId || !riderPrmId || !customerName || !territory || !telecallerName) valid = false;
    if (mobileNumber.length !== 10) valid = false;
    if (!selectedCallStatus) valid = false;
    if (needsResponse(selectedCallStatus) && !selectedResponse) valid = false;

    if (!valid) {
      bannerError.style.display = 'block';
      return;
    }

    if (SCRIPT_URL.includes('PASTE_YOUR')) {
      bannerError.textContent = 'Setup needed: paste your Apps Script Web App URL into SCRIPT_URL in index.html.';
      bannerError.style.display = 'block';
      return;
    }

    submitBtn.disabled = true;
    spinner.style.display = 'inline-block';
    submitLabel.textContent = 'Submitting...';

    const payload = {
      orderId, riderPrmId, mobileNumber, customerName,
      territory, telecallerName,
      callStatus: selectedCallStatus,
      customerResponse: selectedResponse,
      remarks
    };

    try {
      await fetch(SCRIPT_URL, {
        method: 'POST',
        mode: 'no-cors',
        headers: { 'Content-Type': 'text/plain;charset=utf-8' },
        body: JSON.stringify(payload)
      });

      document.getElementById('formCard').style.display = 'none';
      document.getElementById('successScreen').style.display = 'block';
    } catch (err) {
      bannerError.textContent = 'Something went wrong submitting the feedback. Please check your connection and try again.';
      bannerError.style.display = 'block';
    } finally {
      submitBtn.disabled = false;
      spinner.style.display = 'none';
      submitLabel.textContent = 'Submit Feedback';
    }
  });
</script>
